document.addEventListener("DOMContentLoaded", function () {
    "use strict";
  
    // =========================================
    // SKILLVERSE - COMPLETE MAIN SCRIPT
    // =========================================
  
    const STORAGE_KEY = "svCompletedLessons";
    const TOTAL_LESSONS = 40;
    const MILESTONES = [10, 20, 30, 40];
  
    const $ = (selector, parent = document) => parent.querySelector(selector);
    const $$ = (selector, parent = document) => [...parent.querySelectorAll(selector)];
  
    // =========================================
    // STORAGE HELPERS
    // =========================================
  
    function getCompletedLessons() {
      try {
        const data = JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]");
        return Array.isArray(data) ? [...new Set(data)] : [];
      } catch (error) {
        return [];
      }
    }
  
    function saveCompletedLessons(items) {
      localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
      window.dispatchEvent(new Event("svProgressUpdated"));
    }
  
    function escapeHTML(value) {
      return String(value).replace(/[&<>"']/g, character => ({
        "&": "&amp;",
        "<": "&lt;",
        ">": "&gt;",
        '"': "&quot;",
        "'": "&#39;"
      })[character]);
    }
  
    function getCertificateId() {
      let id = localStorage.getItem("svCertificateId");
      if (!id) {
        id = "SV-" + Date.now().toString(36).toUpperCase();
        localStorage.setItem("svCertificateId", id);
      }
      return id;
    }

    // PROFILE NAVIGATION

    const profilePage = $("#profile-page");
    const backButton = $("#back-home");
  
    const homeElements = [
      $("header"),
      $("nav"),
      $("main"),
      $("footer")
    ].filter(Boolean);
  
    function showProfile() {
      homeElements.forEach(element => {
        element.style.display = "none";
      });
  
      if (profilePage) {
        profilePage.hidden = false;
        profilePage.style.display = "block";
      }
  
      window.scrollTo(0, 0);
    }
  
    function showHome() {
      if (profilePage) {
        profilePage.hidden = true;
        profilePage.style.display = "none";
      }
  
      homeElements.forEach(element => {
        element.style.display = "";
      });
  
      window.scrollTo(0, 0);
    }

      profileButton.addEventListener("click", showProfile);
    }

      backButton.addEventListener("click", showHome);

    function displayPhoto(photoData) {
      if (!profilePhoto) return;
      profilePhoto.src = photoData;
      profilePhoto.hidden = false;
      if (photoPlaceholder) {
        photoPlaceholder.hidden = true;
      }

    if (savedPhoto) {
      displayPhoto(savedPhoto);
    }
  
    if (photoUpload) {
      photoUpload.addEventListener("change", function () {
        const file = photoUpload.files[0];
        if (!file) return;
  
        if (!file.type.startsWith("image/")) {
          alert("Please select a valid image.");
          photoUpload.value = "";
          return;
        }
  
        const reader = new FileReader();
        reader.onload = function (event) {
          const image = new Image();
          image.onload = function () {
            const canvas = document.createElement("canvas");
            const maxSize = 400;
            let width = image.width;
            let height = image.height;
  
            if (width > height && width > maxSize) {
              height = (maxSize * height) / width;
              width = maxSize;
            } else if (height > maxSize) {
              width = (maxSize * width) / height;
              height = maxSize;
            }
  
            canvas.width = width;
            canvas.height = height;
            canvas.getContext("2d").drawImage(image, 0, 0, width, height);
  
            try {
              const photoData = canvas.toDataURL("image/jpeg", 0.75);
              localStorage.setItem("svPhoto", photoData);
              displayPhoto(photoData);
              alert("Profile photo uploaded successfully!");
            } catch (error) {
              alert("Photo could not be saved. Choose a smaller image.");
            }
          };
  
          image.onerror = function () {
            alert("Could not open this image.");
          };
  
          image.src = event.target.result;
        };
  
        reader.onerror = function () {
          alert("Could not read this image.");
        };
  
        reader.readAsDataURL(file);
      });
    }

      saveProfileButton.addEventListener("click", function () {
        if (!nameInput || !emailInput || !skillsInput) return;

          alert("Please enter your name.");
          return;

          profileName.textContent = name;
        }

    function updateAchievements() {
      const completed = getCompletedLessons();
      const count = Math.min(completed.length, TOTAL_LESSONS);
      const percentage = Math.round((count / TOTAL_LESSONS) * 100);
  
      const countElement = $("#achievement-count");
      const bar = $("#achievement-bar");
      const message = $("#achievement-message");
  
      if (countElement) {
        countElement.textContent = count + " / " + TOTAL_LESSONS;
      }
  
      if (bar) {
        bar.style.width = percentage + "%";
      }
  
      if (message) {
        message.textContent =
          percentage + "% completed — " +
          (TOTAL_LESSONS - count) + " lessons remaining.";
      }
  
      MILESTONES.forEach(function (milestone) {
        const card = $(`[data-milestone="${milestone}"]`);
        if (!card) return;
  
        const status = $(".badge-status", card);
        if (count >= milestone) {
          card.style.border = "2px solid #00d9ff";
          if (status) {
            status.textContent = "✅ Unlocked";
            status.style.color = "#00d9ff";
          }
        } else {
          card.style.border = "";
          if (status) {
            status.textContent = "🔒 Locked";
            status.style.color = "";
          }
        }
      });
  
      const title = $("#certificate-title");
      const certificateMessage = $("#certificate-message");
      const printButton = $("#print-certificate");
  
      if (count >= TOTAL_LESSONS) {
        if (title) {
          title.textContent = "🏆 Certificate Unlocked!";
        }
        if (certificateMessage) {
          certificateMessage.textContent =
            "Congratulations! You completed all 40 SKILLVERSE lessons.";
        }
        if (printButton) {
          printButton.disabled = false;
        }
      } else {
        if (title) {
          title.textContent = "Certificate Locked";
        }
        if (certificateMessage) {
          certificateMessage.textContent =
            "Complete all 40 lessons to unlock your certificate.";
        }
        if (printButton) {
          printButton.disabled = true;
        }
      }
    }
  
    // =========================================

        } else {
          learnSection.prepend(tracker);

      const tracker = $("#learning-progress-tracker", learnSection);
  
      function updateProgress() {
        const completed = getCompletedLessons();
        const count = Math.min(completed.length, TOTAL_LESSONS);
        const remaining = TOTAL_LESSONS - count;
        const percentage = Math.round((count / TOTAL_LESSONS) * 100);
  
        const countElement = $("#progress-count", tracker);
        const bar = $("#progress-bar", tracker);
        const percent = $("#progress-percent", tracker);
  
        if (countElement) {
          countElement.textContent =
            "Completed: " + count +
            " | Remaining: " + remaining +
            " | Total: " + TOTAL_LESSONS;
        }
  
        if (bar) {
          bar.style.width = percentage + "%";
        }
  
        if (percent) {
          percent.textContent = percentage + "% Completed";
        }
      }
  
      lessons.forEach(function (lesson, index) {
        if ($(".lesson-complete-button", lesson)) return;
  
        const lessonId = "lesson-" + index;
        const button = document.createElement("button");
        button.type = "button";
        button.className = "lesson-complete-button";
        button.style.cssText = `
          display:block;
          margin:12px 0;
          padding:10px 15px;
          border:none;
          border-radius:8px;
          background:#087f9c;
          color:white;
          cursor:pointer;
        `;
  
        function updateButton() {
          const isCompleted = getCompletedLessons().includes(lessonId);
          button.textContent = isCompleted
            ? "✓ Completed — Undo"
            : "✓ Mark as Completed";
          button.style.background = isCompleted ? "#218838" : "#087f9c";
        }
  
        updateButton();
  
        button.addEventListener("click", function (event) {
          event.preventDefault();
          event.stopPropagation();
  
          let completed = getCompletedLessons();
          if (completed.includes(lessonId)) {
            completed = completed.filter(id => id !== lessonId);
          } else {
            completed.push(lessonId);
          }
  
          saveCompletedLessons(completed);
          updateButton();
          updateProgress();
          updateAchievements();
        });
  
        const summary = $("summary", lesson);
        if (summary) {
          summary.insertAdjacentElement("afterend", button);
        } else {
          lesson.appendChild(button);
        }
      });
  
      updateProgress();
    }

      const canvas = $("#sv-certificate-canvas", overlay);
      const ctx = canvas.getContext("2d");
      const printImg = $("#sv-print-img", overlay);
      let isTemplateReady = false;
  
      function drawOverlayDetails() {
        // 1. Student Name Box
        const nameX = 1500 * 0.297;
        const nameY = 1000 * 0.393;
        const nameW = 1500 * 0.422;
        const nameH = 1000 * 0.118;
  
        ctx.fillStyle = "#031126";
        ctx.fillRect(nameX, nameY, nameW, nameH);
  
        ctx.strokeStyle = "#d9ad42";
        ctx.lineWidth = 3;
        ctx.beginPath();
        ctx.moveTo(nameX, nameY + nameH);
        ctx.lineTo(nameX + nameW, nameY + nameH);
        ctx.stroke();
  
        let fontSize = 56;
        ctx.fillStyle = "#f4d47d";
        ctx.textAlign = "center";
        ctx.textBaseline = "middle";
  
        do {
          ctx.font = `italic bold ${fontSize}px Georgia, "Times New Roman", serif`;
          fontSize -= 2;
        } while (
          ctx.measureText(studentName).width > nameW - 30 &&
          fontSize > 20
        );
  
        ctx.fillText(studentName, nameX + nameW / 2, nameY + nameH / 2);
  
        // 2. Issue Date
        const dateX = 1500 * 0.847;
        const dateY = 1000 * 0.6925;
        const dateW = 1500 * 0.122;
        const dateH = 1000 * 0.034;
  
        ctx.fillStyle = "#061326";
        ctx.fillRect(dateX, dateY, dateW, dateH);
        ctx.fillStyle = "#ffffff";
        ctx.font = "bold 15px Arial, sans-serif";
        ctx.textAlign = "left";
        ctx.fillText(issueDate, dateX + 6, dateY + dateH / 2);
  
        // 3. Certificate ID
        const idX = 1500 * 0.847;
        const idY = 1000 * 0.751;
        const idW = 1500 * 0.122;
        const idH = 1000 * 0.035;
  
        ctx.fillStyle = "#061326";
        ctx.fillRect(idX, idY, idW, idH);
        ctx.fillStyle = "#ffffff";
        ctx.font = "bold 14px Arial, sans-serif";
        ctx.fillText(certificateId, idX + 6, idY + idH / 2);
  
        // 4. QR Box Frame
        const qrX = 1500 * 0.806;
        const qrY = 1000 * 0.787;
        const qrW = 1500 * 0.082;
        const qrH = 1000 * 0.12;
  
        ctx.fillStyle = "#ffffff";
        ctx.fillRect(qrX, qrY, qrW, qrH);
        ctx.strokeStyle = "#d7b45b";
        ctx.lineWidth = 3;
        ctx.strokeRect(qrX, qrY, qrW, qrH);
  
        try {
          printImg.src = canvas.toDataURL("image/png");
        } catch (e) {}
      }
  
      const templateImg = new Image();
      templateImg.onload = function () {
        isTemplateReady = true;
        ctx.drawImage(templateImg, 0, 0, 1500, 1000);
        drawOverlayDetails();
  
        const qrImg = new Image();
        qrImg.crossOrigin = "anonymous";
        qrImg.onload = function () {
          const qrX = 1500 * 0.806 + 5;
          const qrY = 1000 * 0.787 + 5;

          "🎉 Maine SKILLVERSE par all 40 lessons complete karke apna Certificate earn kar liya hai! Tum bhi free mein seekho: " +
          window.location.origin;
  
        canvas.toBlob(async function (blob) {
          if (blob && navigator.share && navigator.canShare) {
            const safeFile =
              (studentName.replace(/[^a-z0-9]/gi, "_") || "SKILLVERSE") +
              "_Certificate.png";
            const file = new File([blob], safeFile, { type: "image/png" });
  
            if (navigator.canShare({ files: [file] })) {
              try {
                await navigator.share({
